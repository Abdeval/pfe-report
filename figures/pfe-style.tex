% Stick-figure actor: \actor{<name>}{<position>}{<label>}
\newcommand{\actor}[3]{%
  \begin{scope}[shift={(#2)}, thick, draw=pInk]
    \draw (0,0.42) circle (0.13);
    \draw (0,0.29) -- (0,0.02) (-0.2,0.2) -- (0.2,0.2) (0,0.02) -- (-0.15,-0.22)
      (0,0.02) -- (0.15,-0.22);
    \node[font=\scriptsize\bfseries, below, align=center] (#1) at (0,-0.24) {#3};
    \coordinate (#1-c) at (0,0.1);
  \end{scope}}
